function buildShirtMask(src, k) {
  return new Promise(resolve => {
    const img = new Image();
    img.onload = () => {
      try {
        const W = 800, H = Math.round(img.height * W / img.width);
        const c = document.createElement('canvas'); c.width = W; c.height = H;
        const x = c.getContext('2d');
        x.fillStyle = '#ffffff'; x.fillRect(0, 0, W, H);
        x.drawImage(img, 0, 0, W, H);
        const d = x.getImageData(0, 0, W, H), px = d.data, n = W * H;
        const L = new Float32Array(n);
        let sum = 0, cnt = 0;
        for (let i = 0; i < n; i++) {
          const l = (px[i * 4] + px[i * 4 + 1] + px[i * 4 + 2]) / 3;
          L[i] = l;
          if (l < 60) { sum += l; cnt++; }
        }
        const mean = cnt ? sum / cnt : 10;
        for (let i = 0; i < n; i++) {
          const a = Math.min(1, Math.max(0, (250 - L[i]) / 40));          // hình dáng áo
          const s = Math.min(255, Math.max(120, 238 + (L[i] - mean) * k)); // nếp vải
          const v = a > 0.5 ? s : 238;
          px[i * 4] = px[i * 4 + 1] = px[i * 4 + 2] = v;
          px[i * 4 + 3] = a * 255;
        }
        x.putImageData(d, 0, 0);
        resolve(c.toDataURL('image/png'));
      } catch (e) { resolve(src); }
    };
    img.onerror = () => resolve(src);
    img.src = src;
  });
}

async function submitOrder() {
  const name = document.getElementById('odName').value.trim();
  const phone = document.getElementById('odPhone').value.replace(/[\s.\-]/g, '');
  const address = document.getElementById('odAddress').value.trim();
  const note = document.getElementById('odNote').value.trim();
  const err = document.getElementById('odError');
  if (!name) { err.textContent = 'Vui lòng nhập họ tên.'; return; }
  if (!/^(0|\+84)\d{9,10}$/.test(phone)) { err.textContent = 'Số điện thoại chưa đúng, vui lòng kiểm tra lại.'; return; }
  if (address.length < 8) { err.textContent = 'Vui lòng nhập địa chỉ giao hàng đầy đủ.'; return; }
  err.textContent = '';

  const p = pdProduct;
  const order = {
    code: 'WRU' + String(Math.floor(10000 + Math.random() * 90000)),
    product: p.name, category: CATEGORIES[p.cat].label, size: pdSize, qty: pdQty,
    total: priceNum(p) * pdQty, name, phone, address, note, createdAt: new Date().toISOString()
  };
  const btn = document.getElementById('odSubmit');
  btn.disabled = true; btn.firstElementChild.textContent = 'Đang gửi...';

  let sent = false;
  if (ORDER_ENDPOINT) {
    try {
      const r = await fetch(ORDER_ENDPOINT, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json', 'Accept': 'application/json' },
        body: JSON.stringify({ _subject: 'Đơn hàng ' + order.code, ...order, chi_tiet: orderText(order) })
      });
      sent = r.ok;
    } catch (e) { sent = false; }
  }
  saveOrder(order);
  showOrderDone(order, sent);
}

  const phone = document.getElementById('loginPhone').value.trim();
  const pass = document.getElementById('loginPass').value.trim();
  if (!phone || !pass) { alert('Vui lòng nhập số điện thoại và mật khẩu.'); return; }
  currentUser = { phone: phone, points: 120 };
  points = currentUser.points;
  updateAuthArea();
  closeModal();
  alert(`Chào mừng ${phone}! Bạn có ${points} điểm tích lũy.`);
  document.getElementById('loginPhone').value = '';
  document.getElementById('loginPass').value = '';
}

function logout() {
  currentUser = null;
  points = 0;
  updateAuthArea();
}

function updateAuthArea() {
  const authDiv = document.getElementById('authArea');
  if (currentUser) {
    authDiv.innerHTML = `
      <div class="user-info">
        
        <span>${currentUser.phone}</span>
        <span class="points-badge">${points} điểm</span>
        <a href="#" onclick="logout(); return false;" style="color: var(--red-accent); font-size: 0.8rem; margin-left: 4px;"></a>
      </div>`;
  } else {
    authDiv.innerHTML = `<a href="#" class="login-btn" onclick="openModal(); return false;">Đăng nhập</a>`;
  }
}

function aiDesign() {
  const fonts = [
    "'Space Grotesk', sans-serif", 'Inter, sans-serif', 'Arial, sans-serif',
    "'Courier New', monospace", 'Georgia, serif', "'Brush Script MT', cursive"
  ];
  const icons = ['fa-star', 'fa-heart', 'fa-bolt', 'fa-crown', 'fa-music', 'fa-fire'];
  const texts = ['WEARU', 'OCEAN', 'RED VIBE', 'BLUE SOUL', 'STUDIO 25', 'TỰ TIN', 'WAVE', 'DREAM'];
  const colors = ['#f4f7fb', '#0a5c9e', '#0a3d73', '#1c8ad8', '#d62839', '#1f1f1f'];
  const randomFont = fonts[Math.floor(Math.random() * fonts.length)];
  const randomIcon = icons[Math.floor(Math.random() * icons.length)];
  const randomText = texts[Math.floor(Math.random() * texts.length)];
  const randomColor = colors[Math.floor(Math.random() * colors.length)];

  document.getElementById('fontSelect').value = randomFont;
  document.getElementById('customText').value = randomText;
  document.getElementById('colorSelect').value = randomColor;
  document.querySelectorAll('.icon-option').forEach(o => o.classList.toggle('selected', o.dataset.icon === randomIcon));
  selectedIcon = randomIcon;
  updateLivePreview();

  // Reset vị trí chữ về ngực áo khi AI tạo mới
  const printArea = document.getElementById('printArea');
  printArea.style.left = '50%';
  printArea.style.top = PRINT_TOP[currentStyle] || '30%';
  printArea.style.transform = 'translate(-50%, -50%)';

  alert('🤖 AI đã tạo thiết kế mới cho bạn!');
}

/* ============ KÉO THẢ VÀ DI CHUYỂN CHỮ ============ */
const printArea = document.getElementById('printArea');
let isDragging = false;
let startX, startY, startLeft, startTop;

printArea.addEventListener('mousedown', (e) => {
  if (e.target.closest('input')) return;
  isDragging = true;
  printArea.classList.add('dragging');
  startX = e.clientX;
  startY = e.clientY;

  const rect = printArea.getBoundingClientRect();
  const parentRect = printArea.parentElement.getBoundingClientRect();
  startLeft = rect.left - parentRect.left;
  startTop = rect.top - parentRect.top;

  printArea.style.transform = 'none';
  printArea.style.left = startLeft + 'px';
  printArea.style.top = startTop + 'px';
});

window.addEventListener('mousemove', (e) => {
  if (!isDragging) return;
  const dx = e.clientX - startX;
  const dy = e.clientY - startY;
  printArea.style.left = (startLeft + dx) + 'px';
  printArea.style.top = (startTop + dy) + 'px';
});

window.addEventListener('mouseup', () => {
  if (isDragging) {
    isDragging = false;
    printArea.classList.remove('dragging');
  }
});

// Hỗ trợ cảm ứng trên mobile
printArea.addEventListener('touchstart', (e) => {
  if (e.target.closest('input')) return;
  isDragging = true;
  printArea.classList.add('dragging');
  startX = e.touches[0].clientX;
  startY = e.touches[0].clientY;

  const rect = printArea.getBoundingClientRect();
  const parentRect = printArea.parentElement.getBoundingClientRect();
  startLeft = rect.left - parentRect.left;
  startTop = rect.top - parentRect.top;

  printArea.style.transform = 'none';
  printArea.style.left = startLeft + 'px';
  printArea.style.top = startTop + 'px';
}, { passive: false });

window.addEventListener('touchmove', (e) => {
  if (!isDragging) return;
  e.preventDefault();
  const dx = e.touches[0].clientX - startX;
  const dy = e.touches[0].clientY - startY;
  printArea.style.left = (startLeft + dx) + 'px';
  printArea.style.top = (startTop + dy) + 'px';
}, { passive: false });

window.addEventListener('touchend', () => {
  if (isDragging) {
    isDragging = false;
    printArea.classList.remove('dragging');
  }
});

  steps.style.flexDirection = 'column';
}

/* ============ 3D SHIRT DRAG ============ */
function enable3DDrag(el) {
  let isDown = false, startX = 0, startY = 0, rotY = 0, rotX = 0, currentRotY = 0, currentRotX = 0;

  el.addEventListener('mousedown', e => {
    if (e.target.closest('.print-area')) return;
    isDown = true;
    startX = e.clientX;
    startY = e.clientY;
    el.style.animation = 'none';
    el.style.transition = 'none';
  });

  window.addEventListener('mousemove', e => {
    if (!isDown) return;
    const dx = e.clientX - startX;
    const dy = e.clientY - startY;
    rotY = currentRotY + dx * 0.5;
    rotX = currentRotX - dy * 0.5;
    el.style.transform = `rotateY(${rotY}deg) rotateX(${rotX}deg)`;
  });

  window.addEventListener('mouseup', () => {
    if (!isDown) return;
    isDown = false;
    currentRotY = rotY;
    currentRotX = rotX;
    el.style.transition = 'transform 0.3s ease';
  });

  el.addEventListener('touchstart', e => {
    if (e.target.closest('.print-area')) return;
    isDown = true;
    startX = e.touches[0].clientX;
    startY = e.touches[0].clientY;
    el.style.animation = 'none';
    el.style.transition = 'none';
  }, { passive: true });

  window.addEventListener('touchmove', e => {
    if (!isDown) return;
    const dx = e.touches[0].clientX - startX;
    const dy = e.touches[0].clientY - startY;
    rotY = currentRotY + dx * 0.5;
    rotX = currentRotX - dy * 0.5;
    el.style.transform = `rotateY(${rotY}deg) rotateX(${rotX}deg)`;
  }, { passive: true });

  window.addEventListener('touchend', () => {
    if (!isDown) return;
    isDown = false;
    currentRotY = rotY;
    currentRotX = rotX;
    el.style.transition = 'transform 0.3s ease';
  });
}

/* ============ SCROLL PROGRESS ============ */
window.addEventListener('scroll', () => {
  const scrollTop = window.scrollY;
  const docHeight = document.documentElement.scrollHeight - window.innerHeight;
  const progress = docHeight > 0 ? (scrollTop / docHeight) * 100 : 0;
  document.getElementById('header').style.setProperty('--scroll-progress', progress + '%');
});
